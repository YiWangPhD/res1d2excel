' Structural relationships
BaseElement <|-- SimpleElement : extends
BaseElement <|-- CombinedElement : extends
SimpleElementCollection "1" *-- "many" SimpleElement : contains
ElementCollection "1" *-- "many" BaseElement : contains
CombinedElement "1" *-- "many" BaseElement : uses
BaseElement <-- statistics_calculator : calculates stats

Res1D <|-- Res1DNetwork : extends
Res1D <|-- Res1DRunoff : extends
res1d_extractors *-- Res1D : uses
res1d_extractors ..> SimpleElementCollection : updates

input_xlsx -up-> res1d2excel : uses
input_dataframes -up-> res1d2excel : uses
json_to_dfs -up-> res1d2excel : JSON input

res1d2excel <-right- res1d_extractors : extracts
exporter --> res1d2excel : uses
exporter_xlsx --|> exporter : writes xlsx
main ..> res1d2excel : entry point

utilities_plotly ..> exporter : optional visualization
Res1D ..> utilities : uses
