' Structural relationships
BaseElement <|-- SimpleElement : extends
BaseElement <|-- CombinedElement : extends
SimpleElementCollection ..> SimpleElement : legacy collection
ElementCollection "1" *-- "many" BaseElement : contains
CombinedElement "1" *-- "many" BaseElement : combines
BaseElement <-- statistics_calculator : calculates stats

Res1D <|-- Res1DNetwork : extends
Res1D <|-- Res1DRunoff : extends
Res1DNetwork *-- ElementRef : indexes
ElementRef *-- DataItemRef : quantity refs
res1d_extractors *-- Res1DNetwork : loads network
res1d_extractors *-- Res1DRunoff : loads runoff
res1d_extractors ..> ElementCollection : fills raw series
res1d_extractors ..> CombinedElement : updates calculated series

input_xlsx ..> input_dataframes : templates and combined sheet
input_json ..> input_dataframes : templates and JSON conversion
input_json ..> res1d2excel : discharge quantity rules
input_dataframes ..> ElementCollection : creates collections
input_dataframes ..> SimpleElement : creates simple elements
ElementCollection ..> CombinedElement : creates combined elements

res1d2excel --> input_xlsx : reads xlsx config
res1d2excel --> input_json : reads JSON config
res1d2excel --> input_dataframes : creates dictionaries
res1d2excel --> ElementCollection : creates combined collection
res1d2excel --> res1d_extractors : extracts and updates
res1d2excel --> exporter : builds output dataframes
res1d2excel --> exporter_xlsx : writes xlsx
main ..> res1d2excel : package entry placeholder

utilities_plotly ..> exporter : optional visualization
Res1D ..> utilities : cleans IDs
Res1DNetwork ..> utilities : cleans IDs
Res1DRunoff ..> utilities : cleans IDs
